    \block{Motivation}{
         \begin{itemize}[itemsep=1.1em]
           \item Understanding how, why, and what researchers cite is of broad interest
           \item Extant theory does not easily support quantitative studies
           \item Network representations of citation support uncovering relationships and structural patterns
           \item Simulations under abstract agent-based models complement social theory and mathematical models~\cite{Gilbert1997}
           \item We have developed a series of idealized agent-based models of citation \cite{Chacko2026,Park2026b,Park2026c} where network growth results from agents making and receiving citations based on established patterns in citation data
           \item CantataS adds community-awareness as an additional feature
         \end{itemize}  
         \vspace{5mm}   
    }
    
    \block{Methods}{
\begin{tikzfigure}[The Life of an Agent] \includegraphics[width=0.9\linewidth]{figures/model.pdf} \end{tikzfigure} 

         \begin{itemize}
             \item Agents are nodes that make citations to and receive citations from other nodes
             \item Mid-level modeling approach by sampling from real world data \cite{Harnagel2019}
         \end{itemize}  
         \vspace{5mm}
  }
